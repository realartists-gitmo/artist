# Model production toolchain. NOT a runtime dependency — the shipped `artist`
# binary contains no Python. This produces the ONNX artifact that `rten` loads,
# and later the fine-tuning run that produces the weights in the first place.
#
# Pinned exactly. An ONNX export is only reproducible if the exporter is: a
# torch or transformers bump can silently change which ops a graph decomposes
# into, and `rten` supports ~130 of them.
#
# T5Gemma-2's config declares `transformers_version: 5.0.0.dev0` and a
# `model_type` of `t5gemma2`. The whole 4.x line raises
# `KeyError: 't5gemma2'` from CONFIG_MAPPING — this needs the 5.x line, and
# 4.57 is NOT sufficient despite postdating `t5gemma` (v1, added in 4.53).
#
# Install torch from https://download.pytorch.org/whl/cpu — the default PyPI
# wheel on Linux drags the CUDA runtime in at roughly 10x the size, for a
# machine that runs this on CPU.
torch==2.9.1
transformers==5.14.1
onnx==1.19.0
onnxruntime==1.23.2
safetensors==0.8.0
numpy==2.2.6
sentencepiece==0.2.1
protobuf==6.33.0
accelerate==1.11.0

# torch 2.9 routes `torch.onnx.export` through the dynamo exporter, which
# builds graphs via onnxscript. Not optional despite being undeclared by torch.
#
# MUST be >= 0.7.x. On 0.5.4 the dynamo export dies at the very last step:
#
#   ValueError: The model contains functions. The version conversion pass does
#   not support functions. (_ConvertVersionPassRequiresInline)
#
# `_compat.export_compat` calls `convert_version` unconditionally, so there is
# no flag to skip it and no opset that avoids it. 0.7.1 inlines first and the
# export succeeds. This pin is what stands between us and a KV-cached decoder:
# the dynamo exporter is the only one that can serialise transformers'
# DynamicCache, and it is the only one that works on this model's masks.
onnxscript==0.7.1
